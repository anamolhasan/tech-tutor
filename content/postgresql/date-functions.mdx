<CodeBlock code={"SELECT now()                                    AS now,\n       created_at                              AS ordered_at,\n       AGE(created_at)                         AS age,\n       created_at::date                        AS day,\n       EXTRACT(YEAR FROM created_at)           AS year\nFROM orders\nORDER BY created_at DESC\nLIMIT 3;"} lang={"sql"} />

<CodeBlock code={"SELECT COUNT(*) AS orders_last_7d\nFROM orders\nWHERE created_at >= now() - INTERVAL '7 days';"} lang={"sql"} title={"শেষ ৭ দিন"} />

<CodeBlock code={"SELECT date_trunc('day', created_at) AS day,\n       COUNT(*)                            AS orders\nFROM orders\nGROUP BY day\nORDER BY day;"} lang={"sql"} title={"date_trunc — প্রতিদিনের aggreg"} />
