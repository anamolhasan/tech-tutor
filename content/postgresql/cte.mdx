<CodeBlock code={"WITH user_spend AS (\n  SELECT u.id, u.name, SUM(o.total) AS total_spent\n  FROM users u\n  LEFT JOIN orders o ON o.user_id = u.id\n  GROUP BY u.id, u.name\n)\nSELECT name, COALESCE(total_spent, 0) AS spent\nFROM user_spend\nORDER BY spent DESC\nLIMIT 5;"} lang={"sql"} title={"রেভেনিউ যুক্ত CTE"} />

<CodeBlock code={"WITH stats AS (\n  SELECT product_id, SUM(qty) AS sold\n  FROM order_items\n  GROUP BY product_id\n),\ntop AS (\n  SELECT product_id FROM stats ORDER BY sold DESC LIMIT 3\n)\nSELECT p.name FROM products p\nJOIN top t ON t.product_id = p.id;"} lang={"sql"} title={"একাধিক CTE"} />
