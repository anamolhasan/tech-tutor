## ValidationPipe ব্যবহার

<CodeBlock code={"import { NestFactory } from \"@nestjs/core\";\nimport { ValidationPipe } from \"@nestjs/common\";\nimport { AppModule } from \"./app.module\";\n\nasync function bootstrap() {\n  const app = await NestFactory.create(AppModule);\n\n  // global ValidationPipe — সব DTO auto-validate\n  app.useGlobalPipes(new ValidationPipe({\n    whitelist: true,       // DTO-তে নেই এমন property strip\n    forbidNonWhitelisted: true, // unknown property → 400\n    transform: true,       // type transform\n  }));\n\n  await app.listen(3000);\n}\nbootstrap();"} lang={"typescript"} title={"main.ts — global validation"} />

<CodeBlock code={"import { Body, Controller, Post } from \"@nestjs/common\";\nimport { CreateUserDto } from \"./dto/create-user.dto\";\n\n@Controller(\"users\")\nexport class UsersController {\n  @Post()\n  create(@Body() dto: CreateUserDto) {\n    // dto valid — whitelist applied\n    return dto;\n  }\n}"} lang={"typescript"} title={"controller-এ DTO"} />

## Custom Pipe

<CodeBlock code={"import { PipeTransform, Injectable, BadRequestException } from \"@nestjs/common\";\n\n@Injectable()\nexport class TitleCasePipe implements PipeTransform {\n  transform(value: string): string {\n    if (typeof value !== \"string\") {\n      throw new BadRequestException(\"string দরকার\");\n    }\n    return value\n      .split(\" \")\n      .map((w) => w[0].toUpperCase() + w.slice(1))\n      .join(\" \");\n  }\n}\n\n// ব্যবহার\n@Param(\"name\", TitleCasePipe) name: string"} lang={"typescript"} title={"src/common/pipes/title-case.pipe.ts"} />

<Tip tone={"warning"} title={"Global Pipe setup"} text={"`whitelist: true` মানে DTO-তে না থাকলে property deleted — API predictable থাকে। যত দ্রুত possible validation world-class solver ব্যবহার করুন।"} />
