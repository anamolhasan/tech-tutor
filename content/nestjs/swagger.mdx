## Setup

<CodeBlock code={"npm i @nestjs/swagger"} lang={"bash"} />

<CodeBlock code={"import { NestFactory } from \"@nestjs/core\";\nimport { SwaggerModule, DocumentBuilder } from \"@nestjs/swagger\";\nimport { AppModule } from \"./app.module\";\n\nasync function bootstrap() {\n  const app = await NestFactory.create(AppModule);\n\n  const config = new DocumentBuilder()\n    .setTitle(\"Nest API\")\n    .setDescription(\"Tech Tutor API documentation\")\n    .setVersion(\"1.0\")\n    .addBearerAuth() // JWT authorization\n    .build();\n\n  const document = SwaggerModule.createDocument(app, config);\n  SwaggerModule.setup(\"api\", app, document); // /api-এ docs\n\n  await app.listen(3000);\n}\nbootstrap();"} lang={"typescript"} title={"main.ts"} />

## DTO-তে Swagger Decorator

<CodeBlock code={"import { ApiProperty } from \"@nestjs/swagger\";\nimport { IsEmail, IsString, MinLength } from \"class-validator\";\n\nexport class CreateUserDto {\n  @ApiProperty({ example: \"রহিম\", description: \"User নাম\" })\n  @IsString()\n  @MinLength(2)\n  name: string;\n\n  @ApiProperty({ example: \"rahim@example.com\" })\n  @IsEmail()\n  email: string;\n\n  @ApiProperty({ example: \"secret123\", minLength: 6 })\n  @IsString()\n  password: string;\n}"} lang={"typescript"} title={"create-user.dto.ts"} />

## Controller-Level Decorator

<CodeBlock code={"import { ApiTags, ApiBearerAuth, ApiResponse } from \"@nestjs/swagger\";\n\n@ApiTags(\"users\") // section grouping\n@Controller(\"users\")\nexport class UsersController {\n  @Get(\":id\")\n  @ApiBearerAuth()\n  @ApiResponse({ status: 200, description: \"সফল\", type: User })\n  @ApiResponse({ status: 404, description: \"User নেই\" })\n  findOne(@Param(\"id\") id: string) {\n    return this.usersService.findOne(id);\n  }\n}"} lang={"typescript"} />

## Production-এ Docs

<CodeBlock code={"// production-এ docs বন্ধ রাখা\nconst app = await NestFactory.create(AppModule);\n\nif (process.env.NODE_ENV !== \"production\") {\n  SwaggerModule.setup(\"api\", app, SwaggerModule.createDocument(\n    app,\n    new DocumentBuilder().setTitle(\"API\").build(),\n  ));\n}"} lang={"typescript"} />
