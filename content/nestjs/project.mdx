## Post Service

<CodeBlock code={"import { Injectable, NotFoundException } from \"@nestjs/common\";\nimport { InjectModel } from \"@nestjs/mongoose\";\nimport { Model } from \"mongoose\";\nimport { Post } from \"./schemas/post.schema\";\nimport { CreatePostDto } from \"./dto/create-post.dto\";\nimport { UpdatePostDto } from \"./dto/update-post.dto\";\n\n@Injectable()\nexport class PostsService {\n  constructor(@InjectModel(Post.name) private postModel: Model<Post>) {}\n\n  async create(dto: CreatePostDto, authorId: string) {\n    const post = await this.postModel.create({ ...dto, author: authorId });\n    return post.populate(\"author\", \"name email\");\n  }\n\n  async findAll() {\n    return this.postModel.find().populate(\"author\", \"name email\").sort(\"-createdAt\");\n  }\n\n  async findOne(id: string) {\n    const post = await this.postModel.findById(id).populate(\"author\", \"name\");\n    if (!post) throw new NotFoundException(\"Post পাওয়া যায়নি\");\n    return post;\n  }\n\n  async update(id: string, authorId: string, dto: UpdatePostDto) {\n    const post = await this.postModel.findOneAndUpdate(\n      { _id: id, author: authorId }, // owner only\n      dto,\n      { new: true },\n    );\n    if (!post) throw new NotFoundException(\"Post খুঁজে পাওয়া যায়নি বা অনুমতি নেই\");\n    return post;\n  }\n\n  async remove(id: string, authorId: string) {\n    const post = await this.postModel.findOneAndDelete({ _id: id, author: authorId });\n    if (!post) throw new NotFoundException(\"Post খুঁজে পাওয়া যায়নি\");\n  }\n}"} lang={"typescript"} title={"src/posts/posts.service.ts"} />

## Post Controller (protected)

<CodeBlock code={"import { Body, Controller, Delete, Get, Param, Post, Put, Req, UseGuards } from \"@nestjs/common\";\nimport { ApiBearerAuth, ApiTags } from \"@nestjs/swagger\";\nimport { JwtAuthGuard } from \"../common/guards/jwt-auth.guard\";\nimport { PostsService } from \"./posts.service\";\n\n@ApiTags(\"posts\")\n@Controller(\"posts\")\nexport class PostsController {\n  constructor(private readonly postsService: PostsService) {}\n\n  @Get()\n  findAll() {\n    return this.postsService.findAll();\n  }\n\n  @Get(\":id\")\n  findOne(@Param(\"id\") id: string) {\n    return this.postsService.findOne(id);\n  }\n\n  @Post()\n  @UseGuards(JwtAuthGuard) // author দরকার → protected\n  @ApiBearerAuth()\n  create(@Body() dto: CreatePostDto, @Req() req: any) {\n    return this.postsService.create(dto, req.user.sub);\n  }\n\n  @Put(\":id\")\n  @UseGuards(JwtAuthGuard)\n  update(@Param(\"id\") id: string, @Body() dto: UpdatePostDto, @Req() req) {\n    return this.postsService.update(id, req.user.sub, dto);\n  }\n\n  @Delete(\":id\")\n  @UseGuards(JwtAuthGuard)\n  remove(@Param(\"id\") id: string, @Req() req) {\n    return this.postsService.remove(id, req.user.sub);\n  }\n}"} lang={"typescript"} title={"src/posts/posts.controller.ts"} />
