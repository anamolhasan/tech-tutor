## ConfigService Inject

<CodeBlock code={"import { Injectable } from \"@nestjs/common\";\nimport { ConfigService } from \"@nestjs/config\";\n\n@Injectable()\nexport class DatabaseService {\n  constructor(private configService: ConfigService) {}\n\n  getConfig() {\n    return {\n      port: this.configService.get<number>(\"PORT\", 3000), // default\n      dbUrl: this.configService.get<string>(\"MONGO_URI\"),\n      env: this.configService.get(\"NODE_ENV\"),\n    };\n  }\n}"} lang={"typescript"} />

## Typed Configuration

<CodeBlock code={"export interface AppConfig {\n  port: number;\n  database: { uri: string };\n  jwt: { secret: string; expiresIn: number };\n}\n\nexport default (): AppConfig => ({\n  port: parseInt(process.env.PORT ?? \"3000\", 10),\n  database: {\n    uri: process.env.MONGO_URI ?? \"mongodb://localhost:27017/nestapp\",\n  },\n  jwt: {\n    secret: process.env.JWT_SECRET ?? \"\",\n    expiresIn: parseInt(process.env.JWT_EXPIRES ?? \"3600\", 10),\n  },\n});"} lang={"typescript"} title={"src/config/configuration.ts"} />

<CodeBlock code={"ConfigModule.forRoot({\n  load: [configuration], // typed config\n  isGlobal: true,\n})"} lang={"typescript"} />

## Environment Validation (Joi)

<CodeBlock code={"import * as Joi from \"joi\";\n\nConfigModule.forRoot({\n  validationSchema: Joi.object({\n    PORT: Joi.number().default(3000),\n    MONGO_URI: Joi.string().required(),\n    JWT_SECRET: Joi.string().required(),\n    NODE_ENV: Joi.string()\n      .valid(\"development\", \"production\", \"test\")\n      .default(\"development\"),\n  }),\n  validationOptions: {\n    allowUnknown: true, // অন্য keys বাদ যাবে না\n    abortEarly: false,\n  },\n})"} lang={"typescript"} />
