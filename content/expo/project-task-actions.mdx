<CodeBlock code={"export type Task = {\n  id: string;\n  title: string;\n  done: boolean;\n  createdAt: string;\n};\n\nexport function makeTask(title) {\n  return {\n    id: Date.now().toString(),\n    title,\n    done: false,\n    createdAt: new Date().toISOString(),\n  };\n}\n\nexport function addTask(list, title) {\n  return [makeTask(title), ...list];\n}\n\nexport function toggleTask(list, id) {\n  return list.map((t) => (t.id === id ? { ...t, done: !t.done } : t));\n}\n\nexport function deleteTask(list, id) {\n  return list.filter((t) => t.id !== id);\n}"} lang={"ts"} title={"lib/tasks.ts"} />
