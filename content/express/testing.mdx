## API Test

<CodeBlock code={"const request = require(\"supertest\");\nconst app = require(\"../app\");\n\ndescribe(\"Users API\", () => {\n  test(\"GET /api/users → 200 ও array\", async () => {\n    const res = await request(app).get(\"/api/users\");\n    expect(res.statusCode).toBe(200);\n    expect(Array.isArray(res.body)).toBe(true);\n  });\n\n  test(\"POST /api/users → validation error\", async () => {\n    const res = await request(app)\n      .post(\"/api/users\")\n      .send({ name: \"\" }); // invalid\n\n    expect(res.statusCode).toBe(400);\n  });\n\n  test(\"POST /api/users → 201\", async () => {\n    const res = await request(app)\n      .post(\"/api/users\")\n      .send({ name: \"রহিম\", email: \"r@test.com\" });\n\n    expect(res.statusCode).toBe(201);\n    expect(res.body).toHaveProperty(\"_id\");\n  });\n});"} lang={"javascript"} title={"__tests__/users.test.js"} />

## Unit Test (Middleware)

<CodeBlock code={"const { protect } = require(\"../middleware/auth\");\n\ndescribe(\"Auth middleware\", () => {\n  test(\"token না থাকলে 401\", () => {\n    const req = { headers: {} };\n    const res = { status: jest.fn().mockReturnThis(), \n                  json: jest.fn() };\n    const next = jest.fn();\n\n    protect(req, res, next);\n\n    expect(res.status).toHaveBeenCalledWith(401);\n    expect(next).not.toHaveBeenCalled();\n  });\n});"} lang={"javascript"} title={"test auth middleware"} />
