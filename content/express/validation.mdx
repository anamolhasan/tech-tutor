## Validation Middleware

<CodeBlock code={"const validate = (schema) => (req, res, next) => {\n  const result = schema.safeParse(req.body);\n\n  if (!result.success) {\n    const errors = result.error.issues.map(issue => ({\n      field: issue.path.join(\".\"),\n      message: issue.message,\n    }));\n    return res.status(400).json({ error: \"Validation failed\", errors });\n  }\n\n  req.body = result.data; // cleaned data\n  next();\n};\n\nmodule.exports = validate;"} lang={"javascript"} title={"middleware/validate.js"} />

## Use in Route

<CodeBlock code={"const validate = require(\"../middleware/validate\");\nconst { createUserSchema } = require(\"../validators/userValidator\");\n\nrouter.post(\"/\", validate(createUserSchema), async (req, res) => {\n  const user = await User.create(req.body);\n  res.status(201).json(user);\n});"} lang={"javascript"} />
