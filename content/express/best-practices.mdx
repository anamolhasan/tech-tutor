## 3. Central Error Handler

<CodeBlock code={"// এক জায়গায় সব error\napp.use((err, req, res, next) => {\n  logger.error(err.stack);\n\n  if (err.name === \"ValidationError\") {\n    return res.status(400).json({ error: \"Data invalid\" });\n  }\n  if (err.code === 11000) { // duplicate key\n    return res.status(409).json({ error: \"ডুপ্লিকেট\" });\n  }\n\n  res.status(err.status || 500).json({\n    error: err.message || \"Server error\",\n  });\n});"} lang={"javascript"} />

## 5. Logging

<CodeBlock code={"const winston = require(\"winston\");\n\nconst logger = winston.createLogger({\n  level: process.env.NODE_ENV === \"production\" ? \"info\" : \"debug\",\n  format: winston.format.json(),\n  transports: [\n    new winston.transports.File({ filename: \"logs/error.log\", level: \"error\" }),\n    new winston.transports.File({ filename: \"logs/all.log\" }),\n  ],\n});\n\nif (process.env.NODE_ENV !== \"production\") {\n  logger.add(new winston.transports.Console({ format: winston.format.simple() }));\n}\n\nmodule.exports = logger;"} lang={"javascript"} title={"utils/logger.js"} />
