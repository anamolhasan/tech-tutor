## app.js

<CodeBlock code={"require(\"dotenv\").config();\nconst express = require(\"express\");\nconst cors = require(\"cors\");\nconst helmet = require(\"helmet\");\n\nconst app = express();\n\napp.use(helmet());\napp.use(cors());\napp.use(express.json());\n\napp.get(\"/\", (req, res) => res.json({ message: \"Task API\" }));\n\napp.use(\"/api/auth\", require(\"./routes/auth\"));\napp.use(\"/api/tasks\", require(\"./routes/tasks\"));\n\n// 404\napp.use((req, res) => res.status(404).json({ error: \"Not found\" }));\n\n// error handler\napp.use((err, req, res, next) => {\n  console.error(err);\n  res.status(err.status || 500).json({ error: err.message || \"Server error\" });\n});\n\nmodule.exports = app;"} lang={"javascript"} title={"src/app.js"} />

## models/Task.js

<CodeBlock code={"const mongoose = require(\"mongoose\");\n\nconst taskSchema = new mongoose.Schema({\n  title: { type: String, required: true, trim: true, maxlength: 100 },\n  description: { type: String, default: \"\", maxlength: 500 },\n  completed: { type: Boolean, default: false },\n  priority: {\n    type: String,\n    enum: [\"low\", \"medium\", \"high\"],\n    default: \"medium\",\n  },\n  user: { type: mongoose.Schema.Types.ObjectId, ref: \"User\", required: true },\n  createdAt: { type: Date, default: Date.now },\n});\n\nmodule.exports = mongoose.model(\"Task\", taskSchema);"} lang={"javascript"} title={"src/models/Task.js"} />

## routes/tasks.js

<CodeBlock code={"const express = require(\"express\");\nconst router = express.Router();\nconst Task = require(\"../models/Task\");\nconst { protect } = require(\"../middleware/auth\");\nconst { validateTask } = require(\"../validators/taskValidator\");\n\n// সব middleware এই router-এ\nrouter.use(protect);\n\n// GET /api/tasks\nrouter.get(\"/\", async (req, res, next) => {\n  try {\n    const tasks = await Task.find({ user: req.user._id }).sort(\"-createdAt\");\n    res.json(tasks);\n  } catch (err) { next(err); }\n});\n\n// POST /api/tasks\nrouter.post(\"/\", validateTask, async (req, res, next) => {\n  try {\n    const task = await Task.create({ ...req.body, user: req.user._id });\n    res.status(201).json(task);\n  } catch (err) { next(err); }\n});\n\n// PATCH /api/tasks/:id\nrouter.patch(\"/:id\", async (req, res, next) => {\n  try {\n    const task = await Task.findOneAndUpdate(\n      { _id: req.params.id, user: req.user._id }, // own task only\n      req.body,\n      { new: true, runValidators: true }\n    );\n    if (!task) return res.status(404).json({ error: \"Task খুঁজে পাওয়া যায়নি\" });\n    res.json(task);\n  } catch (err) { next(err); }\n});\n\n// DELETE /api/tasks/:id\nrouter.delete(\"/:id\", async (req, res, next) => {\n  try {\n    const task = await Task.findOneAndDelete({\n      _id: req.params.id,\n      user: req.user._id,\n    });\n    if (!task) return res.status(404).json({ error: \"Task খুঁজে পাওয়া যায়নি\" });\n    res.json({ message: \"Task মুছে ফেলা হয়েছে\" });\n  } catch (err) { next(err); }\n});\n\nmodule.exports = router;"} lang={"javascript"} title={"src/routes/tasks.js"} />
