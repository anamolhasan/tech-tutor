## useRef ও children

<CodeBlock code={"import { useRef } from \"react\";\n\n\nfunction Box({ children }: { children: ReactNode }) {\n  const inputRef = useRef<HTMLInputElement>(null);\n\n  const focus = () => inputRef.current?.focus();\n\n  return (\n    <div onClick={focus}>\n      <input ref={inputRef} />\n      <div>{children}</div>\n    </div>\n  );\n}"} lang={"tsx"} />
