## createSelector

<CodeBlock code={"import { createSelector } from \"@reduxjs/toolkit\";\n\nexport const selectCartItems = (state: RootState) => state.cart.items;\n\nexport const selectCartCount = createSelector(\n  [selectCartItems],\n  (items) => items.reduce((sum, i) => sum + i.qty, 0),\n);\n\nexport const selectCartTotal = createSelector(\n  [selectCartItems],\n  (items) =>\n    items.reduce(\n      (sum, i) => sum + i.qty * (pricePerItem(i.productId) ?? 0),\n      0,\n    ),\n);"} lang={"ts"} title={"src/features/cart/cartSelectors.ts"} />
