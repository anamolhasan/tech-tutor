## ৪. Multi-stage + production deps

<CodeBlock code={"# builder\nFROM node:20-alpine AS builder\n...\nRUN npm ci && npm run build\n\n# runtime — devDependencies ছাড়া\nFROM node:20-alpine\nCOPY --from=builder /app/node_modules ./node_modules\nCOPY --from=builder /app/dist ./dist"} lang={"dockerfile"} title={"প্রোডাকশন deps আলাদা"} />
