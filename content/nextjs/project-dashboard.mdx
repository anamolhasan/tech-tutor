## 2. Dashboard Page

<CodeBlock code={"import { headers } from \"next/headers\";\n\nexport default async function DashboardPage() {\n  const host = headers().get(\"host\");\n  const baseUrl = `http${host?.startsWith(\"localhost\") ? \"\" : \"s\"}://${host}`;\n  const res = await fetch(`${baseUrl}/api/stats`);\n  const stats = await res.json();\n\n  return (\n    <main className=\"p-6\">\n      <h1 className=\"text-2xl font-bold\">Dashboard</h1>\n      <div className=\"mt-6 grid grid-cols-1 gap-4 sm:grid-cols-3\">\n        <div className=\"rounded-xl bg-white p-6 shadow\">\n          <p className=\"text-sm text-gray-500\">Total Users</p>\n          <p className=\"mt-1 text-2xl font-bold\">{stats.users}</p>\n        </div>\n        <div className=\"rounded-xl bg-white p-6 shadow\">\n          <p className=\"text-sm text-gray-500\">Revenue</p>\n          <p className=\"mt-1 text-2xl font-bold\">${stats.revenue}</p>\n        </div>\n        <div className=\"rounded-xl bg-white p-6 shadow\">\n          <p className=\"text-sm text-gray-500\">Orders</p>\n          <p className=\"mt-1 text-2xl font-bold\">{stats.orders}</p>\n        </div>\n      </div>\n    </main>\n  );\n}"} lang={"tsx"} title={"app/dashboard/page.tsx"} />
