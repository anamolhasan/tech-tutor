<InParagraph text={"Edit-ও একই pattern — শুধু `mutationFn`-এ id আর নতুন values যায়। আমরা detail cache-ও আপডেট করব যেন পেজে edit-এর পর সঠিক ডেটা থাকে।"} />

<CodeBlock code={"export function useUpdateProduct() {\n  const queryClient = useQueryClient();\n\n  return useMutation({\n    mutationFn: ({ id, input }: { id: string; input: CreateProductInput }) =>\n      updateProduct(id, input),\n\n    onSuccess: (data) => {\n      // list-গুলো ইনভালিড (table আপডেট)\n      queryClient.invalidateQueries({ queryKey: productKeys.lists() });\n\n      // edit করা product-এর detail cache-ও হালনাগাদ\n      queryClient.setQueryData(\n        productKeys.detail(data.id),\n        data\n      );\n\n      toast.success(\"Product আপডেট হয়েছে\");\n    },\n\n    onError: (error) => toast.error(error.message),\n  });\n}"} lang={"ts"} title={"use-product-mutations.ts (add)"} />

## কেন setQueryData detail-এ

<InParagraph text={"সংখ্যাগত দুই অপরে শুধু `invalidateQueries` করলে detail page আবার fetch করবে — দ্রুত তো ঠিকই, তবে ব্যবহারকারী যদি ঠিক edit-করা item-এ থাকেন, **instant** দেখানোর জন্য আমরা নিজে cache সেট করলাম। দুটোই জায়েজ।"} />
